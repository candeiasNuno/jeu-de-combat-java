package ui;
public class Affichage {
    public static final int LARGEUR = 70;
    private static final int LARGEUR_TEXTE = LARGEUR - 4;

/* AFFICHAGE CLASSIQUE */
    public static void haut(){
        System.out.println("╔" + "═".repeat(LARGEUR - 2) + "╗");
    }
    public static void ligneSimple(){
        System.out.println("─".repeat(LARGEUR));
    }
    public static void bas(){
        System.out.println("╚" + "═".repeat(LARGEUR - 2) + "╝");
    }
    public static void separateur(){
        System.out.println("╠" + "═".repeat(LARGEUR - 2) + "╣");
    }
    public static void separateurDeuxColonnes() {
        System.out.println("╠" + "═".repeat(33) + "╦" + "═".repeat(34) + "╣");
    }
    public static void separateurDeuxColonnesVersHaut() {
        System.out.println("╠" + "═".repeat(33) + "╩" + "═".repeat(34) + "╣");
    }
    public static void ligneVide(){
        System.out.println("╠" + " ".repeat(LARGEUR - 2) + "╣");
    }
    public static void saut(){
        System.out.println("║" + " ".repeat(LARGEUR - 2) + "║");
    }
    public static void clear(){
        System.out.print("\033[H\033[2J"); // Commande pour nettoyer le terminal
        System.out.flush();
    }

    public static void separateurMenuHaut() {
        System.out.println("╠" + "═".repeat(33) + "╦" + "═".repeat(34) + "╣");
    }
    public static void separateurMenuMilieu() {
        System.out.println("╠" + "═".repeat(33) + "╬" + "═".repeat(34) + "╣");
    }
    public static void basMenu() {
        System.out.println("╚" + "═".repeat(33) + "╩" + "═".repeat(34) + "╝");
    }

/* AFFICHAGE AVEC TEXTE */
    public static void ligne(String texte) {
        System.out.printf("║ %-" + LARGEUR_TEXTE + "s ║%n", tronquer(texte, LARGEUR_TEXTE));
    }
    public static void ligneSpecial(String texte1, String texte2, String texte3) {
        // 66 caractères disponibles : 3 colonnes de 20 caractères + 2 écarts de 3 espaces (20 + 3 + 20 + 3 + 20 = 66)
        int largeurCol = (LARGEUR_TEXTE - 6) / 3; // Donne exactement 20

        System.out.printf("║ %-" + largeurCol + "s   %-" + largeurCol + "s   %-" + largeurCol + "s ║%n",
                tronquer(texte1, largeurCol),
                tronquer(texte2, largeurCol),
                tronquer(texte3, largeurCol));
    }
    public static void ligneCentree(String texte) {
        String texteSecurise = tronquer(texte, LARGEUR_TEXTE);
        int espacesGauche = (LARGEUR_TEXTE - texteSecurise.length()) / 2;
        int espacesDroite = LARGEUR_TEXTE - texteSecurise.length() - espacesGauche;

        System.out.println("║ " + " ".repeat(Math.max(0, espacesGauche))
                + texteSecurise
                + " ".repeat(Math.max(0, espacesDroite)) + " ║");
    }
    public static void deuxColonnes(String gauche, String droite) {
        System.out.printf("║ %-31s ║ %-32s ║%n", tronquer(gauche, 31), tronquer(droite, 32));
    }
    public static void ligneVideMenu() {
        // 1 bordure + 33 espaces + 1 bordure + 34 espaces + 1 bordure = 70 caractères
        System.out.println("║" + " ".repeat(33) + "║" + " ".repeat(34) + "║");
    }

/* BARRE VISUELLES DYNAMIQUES */
    public static String genererBarre(int valeurActuelle, int valeurMax) {
        if (valeurMax <= 0) return "[░░░░░░░░░░]";

        int blocsPleins = Math.max(0, Math.min(10, (int) Math.round(((double) valeurActuelle / valeurMax) * 10)));
        int blocsVides = 10 - blocsPleins;

        return "[" + "█".repeat(blocsPleins) + "░".repeat(blocsVides) + "]";
    }
    public static String genererJaugeSpeciale(int jaugeActuelle) {
        int jaugeSecurisee = Math.max(0, Math.min(3, jaugeActuelle));
        return "[" + "■".repeat(jaugeSecurisee) + "□".repeat(3 - jaugeSecurisee) + " ]";
    }

                tronquer(c1, 3),

/* AFFICHAGE VICTOIRE */
    public static void banniereEncadree(String texte) {
        // Largeur totale du cadre intérieur (sur les 66 cases disponibles de LARGEUR_TEXTE)
        int largeurCadre = 52;
        int largeurInterieure = largeurCadre - 2; // 50 caractères entre les barres │...│
        String texteSecurise = tronquer(texte, largeurInterieure - 4);
        // Centrage du texte à l'intérieur du petit cadre
        int espGauche = (largeurInterieure - texteSecurise.length()) / 2;
        int espDroite = largeurInterieure - texteSecurise.length() - espGauche;
        // Assemblage des 5 lignes du cadre
        String bordureHaut = "┌" + "─".repeat(largeurInterieure) + "┐";
        String ligneVide   = "│" + " ".repeat(largeurInterieure) + "│";
        String ligneTexte  = "│" + " ".repeat(espGauche) + texteSecurise + " ".repeat(espDroite) + "│";
        String bordureBas  = "└" + "─".repeat(largeurInterieure) + "┘";
        // Affichage centré dans le grand cadre de 70 caractères
        ligneCentree(bordureHaut);
        ligneCentree(ligneVide);
        ligneCentree(ligneTexte);
        ligneCentree(ligneVide);
        ligneCentree(bordureBas);
    }

/* SECURITE ANTI-DÉBORDEMENTS */
    private static String tronquer(String texte, int max) {
        if (texte == null) return "";
        if (texte.length() > max) {
            return texte.substring(0, max - 3) + "...";
        }
        return texte;
    }
}

package donnees;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardOpenOption;
import java.util.ArrayList;
import java.util.List;
import java.io.IOException;
import combattants.Combattant;
import combattants.Shoto;
import combattants.Assasin;
import combattants.Grappler;
import combattants.shoto.Luke;
import moteur.Roster;
import combattants.Zoner;

public class Sauvegarde {
/* METHODES */
    public void sauvegardeRoster(Roster roster){
        Path chemin = Paths.get("sauvegarde_roster.csv");
        List<String> lignesAEcrire = new ArrayList<>();

        // Transformation des objets en format texte (CSV)
        for (Combattant c : roster.getRoster()){
            String ligne = c.getClass().getSimpleName() + ";" + c.getNom() + ";" + c.getVie() + ";" + c.getAttaqueDeBase() + ";"
                    + c.getVitesse() + ";" + c.getStunMax() + ";" + c.getJaugeSpecial() + ";" + c.getEstVivant();
            lignesAEcrire.add(ligne);
        }

        try {
            // Écrase l'ancien fichier (TRUNCATE) ou le crée (CREATE)
            Files.write(chemin, lignesAEcrire, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
            System.out.println("Roster sauvegardé avec succès !");
        } catch (IOException e){
            System.out.println("Erreur grave lors de la sauvegarde du roster : " + e.getMessage());
        }
    }
    public Roster chargerRoster(){
        Roster ros = new Roster();
        Path chemin = Paths.get("sauvegarde_roster.csv");

        try {
            List<String> lignesLues = Files.readAllLines(chemin);

            for(String ligne : lignesLues){
                // Découpage des statistiques séparées par des points-virgules
                String[] morceaux = ligne.split(";");

                String typeCombattant = morceaux[0];
                String nomCombattant = morceaux[1];
                String vieCombattant = morceaux[2];
                String attaqueDeBaseCombattant = morceaux[3];
                String vitesseCombattant = morceaux[4];
                String stunMax = morceaux[5];
                String jaugeSpecial = morceaux[6];
                String vivant = morceaux[7];

                Combattant combattant = null;

                // Instanciation dynamique selon la classe lue dans le fichier
                switch (typeCombattant){
                    case "Shoto":
                        combattant = new Shoto(nomCombattant);
                        break;
                    case "Assasin":
                        combattant = new Assasin(nomCombattant);
                        break;
                    case "Grappler":
                        combattant = new Grappler(nomCombattant);
                        break;
                    case "Zoner":
                        combattant = new Zoner(nomCombattant);
                        break;
                    case "Luke":
                        combattant = new Luke(Integer.parseInt(vieCombattant), Integer.parseInt(attaqueDeBaseCombattant), Integer.parseInt(vitesseCombattant),
                                Integer.parseInt(stunMax));
                        break;
                    default:
                        System.out.println("Classe inconnue : " + typeCombattant);
                        break;
                }

                if (combattant != null) {
                    ros.ajouterCombattant(combattant);
                }
            }
            System.out.println("Chargement réussi !");

        } catch (IOException e){
            System.out.println("Erreur de lecture : " + e.getMessage());
        }

        return ros;
    }
}

    @Override public void lancerCombatV2(){
        int round = 0;
        String actionRoundPrecedent = "> Tour precedent : Aucun tour encore joue !";
        while(c1.getEstVivant() && c2.getEstVivant()){
            round++;
            Action actJ1 = null;
            Action actJ2 = null;
            actJ1 = this.selectionnerAction(c1, round, actionRoundPrecedent);
            this.pause(1000);
            actJ2 = IA.choisirActionIa(c2);
            c1.desactiveStun();//methode de flemmard mais simple
            c2.desactiveStun();//methode de flemmard mais simple
            actionRoundPrecedent = this.resoudreCroisement(actJ1, c1, actJ2, c2);
        }
        Scanner scanner = new Scanner(System.in);

            this.afficheVictoire2(c1, c2, round, actionRoundPrecedent);
            System.out.print("> Appuyez sur [Entrée] pour retourner au Menu Principal...  ");
            scanner.nextLine();
        } else{

            System.out.print("> Appuyez sur [Entrée] pour retourner au Menu Principal...  ");
            scanner.nextLine();
        }

public class Shoto extends Combattant {

        super(nom, vie, attaqueDeBase, vitesse, stunMax);
    }

/* METHODES */
    @Override public RapportAction lanceAttaqueSpecial1(Combattant cible){
        if (this.jaugeSpecial >= 1 ){
            this.jaugeSpecial --;
            int degats = this.attaqueDeBase * 2;
            int degatsStun = 15; //Capacite Special1
            return new RapportAction(degats, degatsStun, 0, 0, "Projectil",
                    null, null);
        }
        return null;
    }
    @Override public RapportAction lanceAttaqueSpecial2(Combattant cible){
        if (this.jaugeSpecial >= 2 ){
            this.jaugeSpecial -= 2;
            int degats = this.attaqueDeBase * 3;
            this.recevoirJaugeSpecial(1); //capacite Special 2
            return new RapportAction(degats, 0, 0, 0, "rush kick",
                    null, null);
        }
        return null;
    }
    @Override public RapportAction lanceAttaqueUltime(Combattant cible){
        if (this.jaugeSpecial >= 3 ){
            this.jaugeSpecial -= 3;
            int degats = (int) (this.attaqueDeBase * 4.5);
            this.jaugeStun = 0; //capacite ultime
            return new RapportAction(degats, 0, 0, 0, "Super projectil",
                    null, null);
        }
        return null;
    }
    @Override public int estimationDegatInflige(int niveauAtt){
        int estimation;
        switch (niveauAtt){
            case 1:
                estimation = this.attaqueDeBase;
                break;
            case 2:
                estimation = this.attaqueDeBase * 2;
                break;
            case 3:
                estimation = this.attaqueDeBase * 3;
                break;
            case 4:
                estimation = (int) (this.attaqueDeBase * 4.5);
                break;
            default:
                estimation = 0;
                break;
        }
        return (int) (estimation * 1.5); //Passif
    }
    @Override public Combattant copieCombattant(){
        Shoto copie = new Shoto(this.nom);
        return copie;
    }
